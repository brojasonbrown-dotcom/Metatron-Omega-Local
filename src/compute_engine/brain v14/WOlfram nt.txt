memoryStep[state_, memory_, input_, alpha_, gamma_, delta_, lambda_] := 
  Module[{nextState, nextMemory},
    nextState = alpha state + gamma input + delta memory;
    nextMemory = (1 - lambda) memory + lambda state;
    <|"State" -> nextState, "Memory" -> nextMemory|>
  ];

    nextState = alpha state + zeta (predicted - state) + input;
    nextMemory = (1 - lambda) memory + lambda state;
    <|"State" -> nextState, "Memory" -> nextMemory, "Predicted" -> predicted|>
  ];
